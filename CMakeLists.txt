cmake_minimum_required(VERSION 3.16)
project(loggerlib VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

option(LOGGER_BUILD_APP       "Build demo application"   ON)
option(LOGGER_BUILD_COLLECTOR "Build stats collector"    ON)
option(LOGGER_BUILD_TESTS     "Build unit tests"         ON)
option(LOGGER_BUILD_DOCS      "Generate API docs"        OFF)
option(LOGGER_ENABLE_WARNINGS "-Wall -Wextra -Wpedantic" ON)
option(LOGGER_ENABLE_WERROR   "-Werror"                  OFF)

list(APPEND CMAKE_MODULE_PATH ${CMAKE_CURRENT_SOURCE_DIR}/cmake)
include(LoggerHelpers)

set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

set(THREADS_PREFER_PTHREAD_FLAG ON)
find_package(Threads REQUIRED)

if(LOGGER_BUILD_TESTS)
    enable_testing()
endif()
add_subdirectory(common)
add_subdirectory(logger)
if(LOGGER_BUILD_APP)
    add_subdirectory(app)
endif()
if(LOGGER_BUILD_COLLECTOR)
    add_subdirectory(collector)
endif()
if(LOGGER_BUILD_TESTS)
    add_subdirectory(tests)
endif()

if(LOGGER_BUILD_DOCS)
    find_package(Doxygen)
    if(DOXYGEN_FOUND)
        set(DOXYGEN_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/docs)
        set(DOXYGEN_GENERATE_LATEX NO)
        set(DOXYGEN_EXTRACT_ALL NO)
        set(DOXYGEN_WARN_IF_UNDOCUMENTED YES)
        set(DOXYGEN_WARN_AS_ERROR FAIL_ON_WARNINGS)
        set(DOXYGEN_QUIET YES)
        set(DOXYGEN_MACRO_EXPANSION YES)
        set(DOXYGEN_EXPAND_ONLY_PREDEF YES)
        set(DOXYGEN_PREDEFINED LOGGER_EXPORT=)
        set(DOXYGEN_USE_MDFILE_AS_MAINPAGE ${CMAKE_CURRENT_SOURCE_DIR}/README.md)
        doxygen_add_docs(docs logger/include README.md specification.md)
    else()
        message(WARNING "LOGGER_BUILD_DOCS is ON but doxygen was not found")
    endif()
endif()
